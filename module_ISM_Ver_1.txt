Function FeeNoWeekDue(sdate As Date, bdate As Date)

edate = CDate("1/" & Format(bdate, "mm/yyyy"))

If sdate > bdate Then
bdate = sdate
End If

nowd = 0

Do While Month(bdate) = Month(edate)
    If Weekday(bdate) = 1 Then
        nowd = nowd + 1
    End If
    bdate = bdate + 1
Loop

FeeNoWeekDue = nowd

End Function

Function SetReportDate()

On Error GoTo ErrorHandler

    SetReportDate = CDate(InputBox("Please Enter a Date in DD/MM/YYYY Format", "Month Begin Date", CDate("1/" & Format(Now(), "mm/yyyy"))))
    DoCmd.SetWarnings False
    Dim SQL As String
        SQL = "UPDATE tblRepMon " & _
        "SET tblRepMon.RepMonDate = '" & SetReportDate & "' " & _
        "WHERE tblRepMon.RepMonID = 1"
    DoCmd.RunSQL SQL
    DoCmd.SetWarnings True
    Exit Function

ErrorHandler:
    SetReportDate = CDate("1/" & Format(Now(), "mm/yyyy"))
    DoCmd.SetWarnings False
    Dim SQL1 As String
    SQL1 = "UPDATE tblRepMon " & _
        "SET tblRepMon.RepMonDate = '" & SetReportDate & "' " & _
        "WHERE tblRepMon.RepMonID = 1"
    DoCmd.RunSQL SQL1
    DoCmd.SetWarnings True

End Function
