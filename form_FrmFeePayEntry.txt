Private Sub FEAmtDue_AfterUpdate()
If Me!FEFDCode = "E" And Sgn(Me!FEAmtDue) = 1 Or Me!FEFDCode = "R" And Sgn(Me!FEAmtDue) = 1 Or Me!FEFDCode = "W" And Sgn(Me!FEAmtDue) = 1 Then
    Me!FEAmtDue = CCur(Me!FEAmtDue / -1)
ElseIf Me!FEFDCode = "D" And Sgn(Me!FEAmtDue) = -1 Then
    Me!FEAmtDue = CCur(Me!FEAmtDue / -1)
End If
End Sub
Private Sub FEDueDate_AfterUpdate()

End Sub

Private Sub FEFDCode_AfterUpdate()
