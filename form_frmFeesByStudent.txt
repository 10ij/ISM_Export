    Me!FEAmtDue = CCur(Me!FEAmtDue / -1)
ElseIf Me!FEFDCode = "D" And Sgn(Me!FEAmtDue) = -1 Then
    Me!FEAmtDue = CCur(Me!FEAmtDue / -1)
End If
End Sub
